# Cluster size is fixed when the cluster is created. To change it: "minikube delete", then start again.
# 6 CPUs / 6 GB fit inside Docker Desktop's default VM (about 7.4 GB on a 16 GB laptop) only if the
# Docker Compose stack is stopped while minikube runs ("docker compose stop").
# --cni=calico: the default CNI silently ignores the NetworkPolicy objects in k8s\network\ (trust boundary).
# The CNI cannot be changed on an existing cluster: "minikube delete" first.
minikube start --cpus=6 --memory=6g --cni=calico
# The dashboard addon runs extra pods; enable it only when needed

# Ordering: "apply -R" goes through folders alphabetically, so "auth" comes before "base" (the namespace).
# Apply the namespace first, then the Kafka broker (services start without it, but log connection
# warnings until it is ready), then everything else. Run the commands from the repository root.
kubectl apply -f k8s\base\
kubectl apply -f k8s\kafka\
kubectl apply -f k8s\ -R

# Minimal stack: deploy only what is being tested instead of "apply -R", e.g. Catalog <-> Inventory messaging
